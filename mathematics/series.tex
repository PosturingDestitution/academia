arithmetic: $\sum_{k=1}^{n}(a + (k - 1)d)$
\begin{enumerate}
	\item $S_n = a + (a + d) + (a + 2d) + \ldots + (a + (n - 1)d)$
	\item $S_n = (a + (n - 1)d) + (a + (n - 2)d) + \ldots + a$	
	\item $2S_n = [a + (a + (n - 1)d] + [(a + d) + (a + (n - 2)d] + \ldots$
	\item $2S_n = n(2a + (n - 1)d)$
	\item $S_n = \frac{n}{2}(2a + (n - 1)d)$
\end{enumerate}

geometric: $\sum_{k=1}^{n}ar^{k-1}$
\begin{enumerate}
	\item $(S_n - rS_n) = (a + ar + ar^2 + \dots + ar^{n-1}) -  (ar + ar^2 + ar^3 + \ldots + ar^n)$
	\item $S_n - rS_n = a-a^r$
	\item $S_n(1 - r) = a-ar^n$
	\item $S_n = \frac{a - ar^n}{1 - r} = \frac{a(1 - r^n)}{1 - r}  \\ r \neq 1$
\end{enumerate}
